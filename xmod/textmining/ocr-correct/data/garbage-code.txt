int editdistance(string pat, string text)
{
    const int m = pat.size();
    const int n = text.size();

    // The dynamic programming table
    matrix<int> D( m+1, n+1 );

    // Initialize
    D(0,0) = 0;
    for ( int i=1; i<=m; i++ ) D(i,0) = i;
    for ( int j=1; j<=n; j++ ) D(0,j) = 0;

    // Recurrence
    for ( int i=1; i<=m; i++ ) {
        for ( int j=1; j<=n; j++ ) {
            int cost;

            if ( pat[i]==text[j] )
                cost = 0;
            else
                cost = 1;

            D(i,j) = min3( D(i-1,j)+1, D(i,j-1)+1, D(i-1,j-1)+cost);
        }
    }

    return D(m,n);
}
